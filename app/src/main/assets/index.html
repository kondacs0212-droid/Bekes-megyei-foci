function receiveNative(event, raw) { try { if (event === 'error') { $('error').textContent = raw; return; } if (event === 'fcmToken') { fcmToken = String(raw || '').trim(); $('error').textContent = fcmToken ? 'FCM-azonosító sikeresen lekérve.' : 'Üres FCM-azonosító érkezett.'; renderAdmin(); return; } if (event === 'tokenCopied') { $('error').textContent = String(raw || 'Azonosító másolva.'); return; } if (event === 'matchDeleted') { $('error').textContent = 'A mérkőzés sikeresen törölve.'; return; } const v = JSON.parse(raw); if (event === 'matches') { matches = v; if ( pendingDeleteId && !matches.some(m => m.id === pendingDeleteId) ) { cancelDelete(); } } if (event === 'players') { players = v; } if (event === 'auth') { auth = v; isAdmin = !!v.admin; if (!isAdmin) fcmToken = ''; $('status').textContent = !v.signedIn ? 'Nyilvános nézet • belépés nélkül' : (v.admin ? 'Admin: ' : 'Belépve, adminjog nélkül: ') + (v.email || ''); $('login').classList.toggle('hidden', !!v.signedIn); $('logout').classList.toggle('hidden', !v.signedIn); if (v.signedIn) { $('loginPassword').value = ''; } } render(); } catch (e) { $('error').textContent = e.message; } }
